# 队列

## 队列的定义

队列（queue）是一种操作受限的线性数据结构，它只允许在一端进行插入操作（enqueue），在另一端进行删除操作（dequeue），先进先出（First In First Out，FIFO）。

## 实现队列

```ts [顺序队列]
/**
 * 顺序队列
 */
class SeqQueue<T> {
  private elems: T[] = [];
  private front = 0;
  private rear = 0;
  private capacity: number = 0;

  constructor(capacity: number) {
    this.elems = new Array(capacity);
    this.front = 0;
    this.rear = 0;
    this.capacity = capacity;
  }

  enQueue(elem: T) {
    if (this.rear < this.capacity) {
      this.elems[this.rear++] = elem;
    }
  }
  deQueue(): T | undefined {
    if (!this.isEmpty()) {
      const elem = this.elems[this.front++];
      return elem;
    }
  }

  isEmpty(): boolean {
    return this.front === this.rear;
  }
}
```

```ts [环形队列]
// 环形队列主要为了解决假溢出的问题\
class CircularQueue<T> {
  private capacity: number;
  private head: number = 0;
  private tail: number = 0;
  private queue: T[] = [];

  constructor(capacity: number) {
    this.capacity = capacity;
  }

  // 入队
  enqueue(item: T) {
    // 计算下一指针位置
    let next = (this.tail + 1) % this.capacity;
    // 判断队列是否已满
    if (next === this.head) {
      throw new Error('Queue is full');
    }
    // 入队
    this.queue[this.tail] = item;
    // 更新尾指针
    this.tail = next;
  }

  dequeue(): T | undefined {
    // 判断队列是否为空
    if (this.isEmpty()) {
      return undefined;
    }
    // 出队
    let item = this.queue[this.head];
    // 更新头指针
    this.head = (this.head + 1) % this.capacity;
    return item;
  }

  size(): number {
    return (this.tail - this.head + this.capacity) % this.capacity;
  }

  isEmpty(): boolean {
    return this.head === this.tail;
  }
}
```
