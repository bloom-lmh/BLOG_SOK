# 生命周期原理

## 介绍

从本质来讲，是发布订阅模式的一种应用。这些函数通过对应的生命周期注册函数注册到对应的实例上，然后在`patch`特定的阶段进行执行，所谓生命周期注册函数就是以`on`开头的函数，比如：

1. `onBeforeMount`：组件实例被创建，但未挂载到`DOM`上时调用。
2. `onMounted`：组件实例被挂载到`DOM`上时调用。
3. `onBeforeUpdate`：组件实例更新之前调用。
4. `onUpdated`：组件实例更新之后调用。
5. `onBeforeUnmount`：组件实例销毁之前调用。
6. `onUnmounted`：组件实例销毁之后调用。
7. `onActivated`：组件实例被激活时调用。
8. `onDeactivated`：组件实例被失活时调用。

## 示例

这里只演示 `onBeforMount` 、`mounted` 、`onBeforeUpdate` 和 `updated` 四个阶段的钩子。

在组件首次挂载的时候会调用`setup`函数，从而触发注册生命周期钩子函数

```js
const ChildComponent = {
  setup(props, { emit, attrs, expose, slots }) {
    onBeforeMount(() => {
      console.log(getCurrentInstance());
      console.log('child component onBeforeMount');
    });
    onMounted(() => {
      console.log('child component onMounted');
    });
    onBeforeUpdate(() => {
      console.log('child component onBeforeUpdate');
    });
    onUpdated(() => {
      console.log('child component onUpdated');
    });
    return () => {
      return h('div', {}, 'child component');
    };
  },
};
const ParentComponent = {
  setup(props, { emit, attrs, expose, slots }) {
    const name = ref('parent');
    onBeforeMount(() => {
      console.log(getCurrentInstance());
      console.log('parent component onBeforeMount');
    });
    onMounted(() => {
      console.log('parent component onMounted');
    });
    onBeforeUpdate(() => {
      console.log('parent component onBeforeUpdate');
    });
    onUpdated(() => {
      console.log('parent component onUpdated');
    });
    setTimeout(() => {
      name.value = 'new parent';
    }, 1000);
    return () => {
      return h(Fragment, {}, [h('div', {}, `${name.value}component`), h(ChildComponent, { value: name.value })]);
    };
  },
};
render(
  // 组件被挂载到一个虚拟节点的type上
  h(ParentComponent, {}),
  app,
);
```

## 源码解析

### 生命周期钩子注册器

我们来看下这些注册函数是怎么去实现的：

```js
import {
  currentInstance,
  setCurrentInstance,
  unsetCurrentInstance,
} from "./component";

/**
 * 生命周期类型
 */
export const enum Lifecycle {
  BEFORE_MOUNT = "bm",
  MOUNTED = "m",
  BEFORE_UPDATE = "bu",
  UPDATED = "u",
}
/**
 * 钩子函数注册器工厂
 * @type {Lifecycle} type 生命周期的类型，用于区分实例上的不同生命周期钩子数组
 */
function createHook(type: Lifecycle) {
  // 利用闭包将当前的实例存在此钩子中
  /**
   * 钩子函数注册器
   * @param hook 钩子函数
   * @param target 当前组件实例
   */
  return (hook, target = currentInstance) => {
    // 实例存在才能在实例上挂载钩子函数
    if (target) {
      // 根据类型获取对应的钩子数组
      const hooks = target[type] || (target[type] = []);
      // 绑定实例到钩子函数上
      const wrapHook = () => {
        setCurrentInstance(target);
        hook.call(target);
        unsetCurrentInstance();
      };
      // 注册钩子函数（发布订阅）
      hooks.push(wrapHook);
    }
  };
}

// 通过函数工厂返回不同的注册钩子注册器
export const onBeforeMount = createHook(Lifecycle.BEFORE_MOUNT);
export const onMounted = createHook(Lifecycle.MOUNTED);
export const onBeforeUpdate = createHook(Lifecycle.BEFORE_UPDATE);
export const onUpdated = createHook(Lifecycle.UPDATED);

// 发布执行订阅函数
export function invokeArrayFns(fns) {
  fns.forEach((fn) => fn());
}

```

```js {8-10,16-18,26-29,35-38}
function setupRenderEffect(instance, container, anchor, parentComponent) {
  const componentUpdate = () => {
    // 更新函数
    // 我们要区分是第一次还是之后的更新，不然会一直叠在上面一直挂载
    const { bm, m, bu, u } = instance; // 拿到生命周期钩子
    if (!instance.isMounted) {
      // 执行挂载前的钩子
      if (bm) {
        invokeArrayFns(bm);
      }
      const subTree = renderComponent(instance); // 生成subTree，由于内部使用了
      patch(null, subTree, container, anchor, instance); // 向下走一层，实现对s
      instance.isMounted = true;
      instance.subTree = subTree;
      // 执行挂载后的钩子
      if (m) {
        invokeArrayFns(m);
      }
    } else {
      const { next } = instance;
      if (next) {
        // 分开两边写更新实在太变态，这里通过next来判断是否为属性或插槽更新
        // 更新属性和插槽
        updateComponentPreRender(instance, next);
      }
      // 执行更新前的钩子
      if (bu) {
        invokeArrayFns(bu);
      }
      // 基于状态的组件更新
      const subTree = renderComponent(instance);
      // 上一次的subTree和此次进行更新
      patch(instance.subTree, subTree, container, anchor, instance);
      instance.subTree = subTree;
      // 执行更新后的钩子
      if (u) {
        invokeArrayFns(u);
      }
    }
  };
  // 这里不直接使用update作为scheduler，而是再包装实现批处理
  // 如果不想做批处理直接使用watchEffect，那么每次更新都会执行scheduler，这样每
  const effect = new ReactiveEffect(componentUpdate, () => queueJob(update));
  const update = (instance.update = () => effect.run());
  update();
}
```
